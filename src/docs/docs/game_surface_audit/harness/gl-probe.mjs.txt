for (const args of [[], ['--use-gl=angle', '--use-angle=swiftshader', '--enable-unsafe-swiftshader'], ['--use-angle=metal']]) {
  const b = await chromium.launch({ headless: true, args });
  const p = await b.newPage();

  await p.waitForFunction(() => window.__idleVillageTestHooks, null, { timeout: 60000 }).catch(() => {});
