const sample = () => page.evaluate((pid) => {

  const w = document.querySelector('[data-testid="quest-window"]');
  const pb = w?.querySelector('[role="progressbar"]');

  const txt = (w?.innerText ?? '').split('\n');
  return { t: Math.round(performance.now()), tick: clk?.currentTick, node: r?.nodeId, status: r?.frontier?.status, readyAt: r?.frontier?.readyAt, ended: r?.ended, outcome: r?.outcome,
    bar: pb ? Number(pb.getAttribute('aria-valuenow')) : null, barLabel: pb?.getAttribute('aria-label') ?? null,

const theatreMotion = async (label) => { // px cambiati nel teatro su 3 s, in 4 campioni
  const box = await page.getByTestId('quest-theatre-frame').boundingBox().catch(() => null); if (!box) return null;
  const shots = []; for (let i = 0; i < 4; i++) { shots.push((await page.screenshot({ clip: box })).toString('base64')); await page.waitForTimeout(1000); }
  const res = await page.evaluate(async (arr) => {
    const load = async (b) => { const im = await createImageBitmap(await (await fetch('data:image/png;base64,' + b)).blob()); const c = new OffscreenCanvas(im.width, im.height); const g = c.getContext('2d'); g.drawImage(im, 0, 0); return g.getImageData(0, 0, im.width, im.height).data; };
    const ds = await Promise.all(arr.map(load)); const out = [];
    for (let k = 1; k < ds.length; k++) { let n = 0; const a = ds[k - 1], b = ds[k]; for (let i = 0; i < a.length; i += 4) if (Math.abs(a[i] - b[i]) + Math.abs(a[i + 1] - b[i + 1]) + Math.abs(a[i + 2] - b[i + 2]) > 24) n++; out.push(+(100 * n / (a.length / 4)).toFixed(2)); }
    return out;
  }, shots);
  return { label, pctChangedPerSecond: res };
};
