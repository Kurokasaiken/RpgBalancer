// drag SOLO MOUSE (nessun hook), con frame a metà percorso
async function mouseDrag(rid, slotId, name) {
  const card = page.locator(`[data-worker-id="${rid}"]`).first();
  const slot = page.getByTestId(`slot-button-${slotId}`).first().or(page.locator(`[data-slot-id="${slotId}"]`).first());
  await card.scrollIntoViewIfNeeded(); const a = await card.boundingBox(); const b = await slot.first().boundingBox();
  if (!a || !b) return { ok: false, why: 'no box' };
  const from = { x: a.x + a.width / 2, y: a.y + a.height / 2 }, to = { x: b.x + b.width / 2, y: b.y + b.height / 2 };
  await page.mouse.move(from.x, from.y); await page.mouse.down();
  await page.mouse.move(from.x + 12, from.y + 6, { steps: 3 });
  const N = 14; for (let i = 1; i <= N; i++) await page.mouse.move(from.x + (to.x - from.x) * i / N, from.y + (to.y - from.y) * i / N, { steps: 2 });
  await page.mouse.move(to.x + 1, to.y); await page.mouse.move(to.x, to.y); await page.waitForTimeout(250);
  if (name) await snap(ctx, name);
  const dropState = await page.locator(`[data-slot-id="${slotId}"][data-drop-state]`).first().getAttribute('data-drop-state').catch(() => null);
  await page.mouse.up(); await page.waitForTimeout(500);
  const asg = await assignments();
  return { ok: asg[slotId] === rid, dropState, assigned: asg[slotId] ?? null };
}
