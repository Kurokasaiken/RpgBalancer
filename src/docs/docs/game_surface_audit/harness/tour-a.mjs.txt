import fs from 'node:fs';
import { openGame, POI, snap, close, lensAudit, warmUp } from './harness.mjs';
const W = Number(process.env.W || 1440), H = Number(process.env.H || 900), LANG = process.env.LANG_IT === '1';
const outDir = new URL(`./out/tourA-${W}x${H}${LANG ? '-it' : ''}/`, import.meta.url).pathname;
const ctx = await openGame({ outDir, viewport: { width: W, height: H } });
const { page } = ctx; const index = [];
const state = async (name, note = '') => { await page.waitForTimeout(150); await snap(ctx, name); const lens = await lensAudit(ctx).catch((e) => [{ error: String(e) }]); fs.writeFileSync(`${outDir}${name}.lens.json`, JSON.stringify(lens, null, 1)); index.push({ name, note, surfaces: lens.map((l) => l.name), issues: lens.reduce((a, l) => a + (l.L1n || 0) + (l.L2_truncated?.length || 0) + (l.L7_circleWithSymbol || 0) + (l.L8n || 0) + (l.L9_tightToFrame || 0), 0) }); };
const tryDo = async (f) => { try { await f(); } catch (e) { index.push({ name: 'ERR', note: String(e).slice(0, 160) }); } };
await page.waitForFunction(() => !document.querySelector('[data-testid="game-boot-screen"]'), null, { timeout: 60000 }).catch(() => {});
await page.waitForTimeout(1500);
index.push({ name: 'warmup', note: String(await warmUp(ctx)) });
if (LANG) { await page.getByRole('button', { name: /^menu$/i }).first().click(); await page.waitForTimeout(400); await page.getByText(/^italiano$/i).first().click().catch(() => {}); await page.waitForTimeout(800); await page.keyboard.press('Escape'); await page.waitForTimeout(400); }
await state('A01-map');
// regioni
for (const [x, y, n] of [[0.25, 0.42, 'west-forest'], [0.62, 0.55, 'east-badlands'], [0.45, 0.62, 'center']]) { await tryDo(async () => { await page.mouse.move(W * x, H * y, { steps: 5 }); await page.waitForTimeout(1100); await state(`A02-region-${n}`); }); }
// POI hover + zoom
await tryDo(async () => { const m = page.locator(`[data-quest-poi-id="${POI}"]`).first(); await m.hover(); await page.waitForTimeout(900); await state('A03-poi-hover'); const b = await m.boundingBox(); await page.screenshot({ path: outDir + 'A03-poi-zoom.png', clip: { x: b.x - 60, y: b.y - 60, width: b.width + 120, height: b.height + 120 } }); });
// planner (POI detail con preview)
await tryDo(async () => { await page.locator(`[data-quest-poi-id="${POI}"]`).first().click(); await page.locator(`[data-testid="floating-panel-quest-expedition-${POI}"]`).waitFor({ timeout: 15000 }); await page.waitForTimeout(900); await state('A04-planner-empty'); });
await tryDo(async () => { await page.locator('[data-worker-id="hero-salvatrice"]').first().hover(); await page.waitForTimeout(1200); await state('A05-planner-herosheet-hover'); await page.mouse.move(W / 2, H - 30); await page.waitForTimeout(500); });
// inventario: tray oggetti e compagni, magazzino
await tryDo(async () => { await page.getByTestId('bag-slot-0').locator('button').click(); await page.waitForTimeout(700); await state('A06-tray-items'); const t = page.locator('[data-testid^="picker-item-"]').first(); await t.hover(); await page.waitForTimeout(900); await state('A07-tray-item-hover'); await t.click(); await page.waitForTimeout(400); await state('A08-tray-item-picked'); await page.locator('[data-testid^="quest-tray-mode-residents"]').click().catch(() => {}); await page.waitForTimeout(600); await state('A09-tray-residents'); await page.keyboard.press('Escape'); await page.waitForTimeout(300); });
await tryDo(async () => { await page.getByTestId('quest-open-stash').click(); await page.waitForTimeout(800); await state('A10-planner-stash'); });
// party e forecast pronto (hook dichiarato: lo stato del planner pieno è già stato provato con input reali in F3)
await tryDo(async () => { for (const [s, r] of [['leader', 'villager-mastro-beppe'], ['member-1', 'hero-sir-spaccaculi'], ['member-2', 'villager-lisetta'], ['bodyguard', 'villager-baldassarre']]) await page.evaluate(([pid, sl, rid]) => window.__idleVillageTestHooks.expedition[pid].assignToSlot(`${pid}:goblin-slot-${sl}`, rid), [POI, s, r]); ctx.steps.push({ step: 'party', how: 'hook-assisted:assignToSlot' }); await page.waitForTimeout(1500); await state('A11-planner-ready'); const tok = page.getByTestId(`slot-token-${POI}:goblin-slot-leader`); await tok.hover(); await page.waitForTimeout(900); await state('A12-slot-hover'); });
await tryDo(async () => { await page.getByTestId(`floating-panel-close-quest-expedition-${POI}`).click(); await page.waitForTimeout(700); });
// roster + scheda personaggio
await tryDo(async () => { if (!(await page.locator('[data-hud-panel="roster"]').count())) await page.keyboard.press('r'); await page.waitForTimeout(700); await state('A13-roster'); await page.locator('[data-worker-id="hero-sir-spaccaculi"]').first().hover(); await page.waitForTimeout(1200); await state('A14-herosheet'); const pin = page.locator('button[aria-label*="in" i][aria-pressed]').first(); await page.mouse.move(W / 2, H - 30); await page.keyboard.press('p'); await page.waitForTimeout(900); await state('A15-herosheet-key-p'); });
// stash, mercante, eventi, menu
await tryDo(async () => { await page.keyboard.press('r'); await page.keyboard.press('v'); await page.waitForTimeout(800); await state('A16-stash'); await page.keyboard.press('v'); await page.keyboard.press('m'); await page.waitForTimeout(800); await state('A17-merchant'); await page.locator('[data-hud-panel="merchant"] button').filter({ hasText: /buy|compra/i }).first().hover().catch(() => {}); await page.waitForTimeout(800); await state('A18-merchant-buy-hover'); await page.keyboard.press('m'); await page.keyboard.press('e'); await page.waitForTimeout(700); await state('A19-events'); await page.keyboard.press('e'); });
await tryDo(async () => { await page.getByRole('button', { name: /^menu$/i }).first().click(); await page.waitForTimeout(600); await state('A20-menu'); const all = page.getByText(/all settings|tutte le impostazioni|impostazioni/i).first(); if (await all.count()) { await all.click(); await page.waitForTimeout(900); await state('A21-settings'); await page.keyboard.press('Escape'); await page.waitForTimeout(400); } });
await tryDo(async () => { await page.getByRole('button', { name: /^menu$/i }).first().click(); await page.waitForTimeout(400); const load = page.getByText(/load game|carica/i).first(); if (await load.count()) { await load.click(); await page.waitForTimeout(900); await state('A22-load-game'); await page.keyboard.press('Escape'); await page.waitForTimeout(400); } });
// villaggio
await tryDo(async () => { await page.keyboard.press('Escape'); await page.locator('[data-nav-id="village"]').click(); for (const t of [300, 900, 1800, 3000]) { await page.waitForTimeout(t === 300 ? 300 : t - [300, 900, 1800, 3000][[300, 900, 1800, 3000].indexOf(t) - 1]); await snap(ctx, `A23-village-enter-${t}ms`); } await state('A24-village'); const wh = page.locator('[data-building-pin="warehouse"]'); await wh.hover(); await page.waitForTimeout(800); await state('A25-village-warehouse-tooltip'); await page.getByTestId('village-registry-chip').click(); await page.waitForTimeout(900); await state('A26-registry'); await page.keyboard.press('Escape'); await page.waitForTimeout(500); notesRegistryAfterEsc = await page.getByTestId('village-registry-list').isVisible().catch(() => null); });
var notesRegistryAfterEsc;
await tryDo(async () => { await page.locator('[data-nav-id="workshop"]').click(); await page.waitForTimeout(900); await state('A27-workshop-click'); index.push({ name: 'workshop-hud-hidden', note: String(await page.evaluate(() => !!document.querySelector('[data-hud-hidden]'))) }); await page.keyboard.press('h').catch(() => {}); await page.locator('[data-nav-id="map"]').click().catch(() => {}); await page.waitForTimeout(1200); });
// invasione (hook di test, dichiarato)
await tryDo(async () => { await page.evaluate(() => window.__idleVillageTestHooks?.startInvasion?.()); ctx.steps.push({ step: 'invasion', how: 'hook-assisted:startInvasion' }); for (const t of [300, 1200]) { await page.waitForTimeout(t === 300 ? 300 : 900); await snap(ctx, `A28-invasion-${t}ms`); } const card = page.getByTestId('world-event-card-body'); await card.waitFor({ timeout: 10000 }); await page.waitForTimeout(600); await state('A29-invasion-card'); const btn = card.locator('button').last(); await btn.click(); for (const t of [300, 900, 1500, 2400, 3600]) { await page.waitForTimeout(t === 300 ? 300 : 600); await snap(ctx, `A30-invasion-peel-${t}ms`); } await page.waitForTimeout(1500); await state('A31-invasion-after'); });
fs.writeFileSync(outDir + 'index.json', JSON.stringify({ index, registryVisibleAfterEsc: notesRegistryAfterEsc, errors: ctx.consoleErrors.slice(0, 5) }, null, 1));
console.log(JSON.stringify(index.map((i) => [i.name, i.issues ?? i.note])));
await close(ctx);
